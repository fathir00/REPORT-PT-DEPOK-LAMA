feePerSession = 40000; } else if (conduct <= 149) { code = "VB3"; feePerSession = 50000; } else { code = "VB4"; feePerSession = 55000; } } else if (points <= 2699) { if (conduct < 70) { code = "WB0"; feePerSession = 0; } else if (conduct <= 99) { code = "WB1"; feePerSession = 40000; } else if (conduct <= 124) { code = "WB2"; feePerSession = 50000; } else if (conduct <= 145) { code = "WB3"; feePerSession = 60000;
